cmake_minimum_required(VERSION 3.21)
project(key_native_checks LANGUAGES C)
set(CMAKE_C_STANDARD 17)
get_filename_component(repo "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(RP2350_MBEDTLS_PATH "" CACHE PATH "Reviewed Mbed TLS 3.6.7 source tree, or fetch pinned sources")
add_subdirectory("${repo}" portable)
include("${repo}/firmware/rp2350/cmake/crypto.cmake")
include("${repo}/firmware/rp2350/cmake/storage.cmake")
if(MINGW)
    # The Windows CRT declaration is required with NO_STD_FUNCTIONS.
    target_compile_options(rp2350_mbedcrypto PRIVATE -include stdio.h)
endif()
add_executable(storage_check storage_check.c)
target_link_libraries(storage_check PRIVATE rp2350_storage_codec)
target_include_directories(storage_check PRIVATE "${repo}/firmware/rp2350/vendor/monocypher")
add_executable(crypto_check crypto_check.c)
target_sources(crypto_check PRIVATE "${repo}/firmware/rp2350/entropy.c")
target_link_libraries(crypto_check PRIVATE rp2350_crypto)
foreach(check IN ITEMS storage_check crypto_check)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${check} PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    endif()
endforeach()
